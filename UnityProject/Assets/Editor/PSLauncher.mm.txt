#import <UIKit/UIKit.h>
#import <TargetConditionals.h>
#import <errno.h>
#import <sys/mman.h>
#import <sys/sysctl.h>
#import <unistd.h>
#import "UnityAppController.h"

enum PSRuntimePreference {
    PSRuntimeAutomatic = 0,
    PSRuntimeInterpreter = 1,
    PSRuntimeJit = 2,
};

enum PSJitMemoryStatus {
    PSJitMemoryNotChecked = 0,
    PSJitMemoryMapped = 1,
    PSJitMemoryUnavailable = 2,
    PSJitMemoryUnsupported = 3,
};

static NSString *const PSRuntimePreferenceKey = @"port.launcher.preferredMode";
static PSJitMemoryStatus PSJitMemory = PSJitMemoryNotChecked;
static int PSJitMemoryError;

static PSRuntimePreference PSPreferredRuntime(void) {
    NSInteger stored = [NSUserDefaults.standardUserDefaults integerForKey:PSRuntimePreferenceKey];
    if (stored == PSRuntimeInterpreter || stored == PSRuntimeJit) return (PSRuntimePreference)stored;
    return PSRuntimeAutomatic;
}

static void PSRefreshJitMemory(void) {
    PSJitMemory = PSJitMemoryNotChecked;
    PSJitMemoryError = 0;
    if (PSPreferredRuntime() == PSRuntimeInterpreter) return;
#if defined(PROSTO_INTERP) && defined(MAP_JIT) && defined(__arm64__) && !TARGET_OS_SIMULATOR
    size_t size = (size_t)getpagesize();
    void *page = mmap(NULL, size, PROT_READ | PROT_WRITE | PROT_EXEC, MAP_PRIVATE | MAP_ANON | MAP_JIT, -1, 0);
    if (page == MAP_FAILED) {
        PSJitMemoryError = errno;
        PSJitMemory = PSJitMemoryUnavailable;
        return;
    }
    if (munmap(page, size) != 0) {
        PSJitMemoryError = errno;
        PSJitMemory = PSJitMemoryUnavailable;
        return;
    }
    // Mapping alone does not verify that a JIT backend can write and execute code.
    PSJitMemory = PSJitMemoryMapped;
#else
    PSJitMemory = PSJitMemoryUnsupported;
#endif
}

// Called on Unity's main thread. The launcher takes the final snapshot before handing off.
extern "C" void PSLauncherReadRuntimeStatus(int *preferred, int *jitMemory, int *jitError) {
    *preferred = PSPreferredRuntime();
    *jitMemory = PSJitMemory;
    *jitError = PSJitMemoryError;
}

static NSString *PSModKey(NSString *mod) { return [@"port.launcher.mod." stringByAppendingString:mod]; }
extern "C" int PSLauncherModState(const char *identifier) {
    NSNumber *value = [NSUserDefaults.standardUserDefaults objectForKey:PSModKey(@(identifier))];
    return value ? value.intValue : -1;
}
extern "C" void PSLauncherMirrorMod(const char *identifier, int enabled) {
    [NSUserDefaults.standardUserDefaults setInteger:enabled forKey:PSModKey(@(identifier))];
}

extern "C" int PSLauncherProbeEnabled(void) {
    return 0;
}

static NSURL *PSZipInbox(void) {
    NSURL *docs = [NSFileManager.defaultManager URLsForDirectory:NSDocumentDirectory inDomains:NSUserDomainMask].firstObject;
    return [docs URLByAppendingPathComponent:@"ModImports" isDirectory:YES];
}

static NSString *PSAttachStatus(void) {
    int query[] = {CTL_KERN, KERN_PROC, KERN_PROC_PID, getpid()};
    struct kinfo_proc info = {};
    size_t size = sizeof(info);
    if (sysctl(query, 4, &info, &size, NULL, 0) || size != sizeof(info)) return @"无法读取";
    return (info.kp_proc.p_flag & P_TRACED) ? @"已连接" : @"未检测到连接";
}

static BOOL PSHasZipExtension(NSURL *url) {
    return [url.pathExtension.lowercaseString isEqualToString:@"zip"];
}

@interface PSLaunchSettings : UITableViewController <UIDocumentPickerDelegate>
@property(nonatomic, copy) void (^play)(void);
@property(nonatomic, copy) NSArray<NSURL *> *mods;
@end

@implementation PSLaunchSettings
- (void)viewDidLoad {
    [super viewDidLoad];
    self.title = @"深空当铺 · 启动设置";
    self.navigationItem.rightBarButtonItem = [[UIBarButtonItem alloc] initWithTitle:@"进入游戏" style:UIBarButtonItemStyleDone target:self action:@selector(startGame)];
    [NSNotificationCenter.defaultCenter addObserver:self selector:@selector(refreshOnActive:) name:UIApplicationDidBecomeActiveNotification object:nil];
    [self refresh];
}
- (void)startGame { if (self.play) self.play(); }
- (void)refreshOnActive:(NSNotification *)note { [self refresh]; }
- (void)refresh {
    PSRefreshJitMemory();
    NSArray *queued = [NSFileManager.defaultManager contentsOfDirectoryAtURL:PSZipInbox() includingPropertiesForKeys:@[NSURLIsRegularFileKey] options:NSDirectoryEnumerationSkipsHiddenFiles error:NULL];
    NSMutableArray<NSURL *> *list = [NSMutableArray array];
    for (NSURL *url in queued) {
        if (!PSHasZipExtension(url)) continue;
        NSNumber *regular = nil;
        [url getResourceValue:&regular forKey:NSURLIsRegularFileKey error:NULL];
        if (regular.boolValue) [list addObject:url];
    }
    self.mods = [list sortedArrayUsingComparator:^NSComparisonResult(NSURL *a, NSURL *b) {return [a.lastPathComponent localizedStandardCompare:b.lastPathComponent];}];
    [self.tableView reloadData];
}
- (NSInteger)numberOfSectionsInTableView:(UITableView *)table {return 3;}
- (NSInteger)tableView:(UITableView *)table numberOfRowsInSection:(NSInteger)section {
    return section == 0 ? 4 : section == 1 ? 2 : 1 + self.mods.count;
}
- (NSString *)tableView:(UITableView *)table titleForHeaderInSection:(NSInteger)section {
    return @[@"运行模式",@"内置模组",@"外部模组"][section];
}
- (NSString *)tableView:(UITableView *)table titleForFooterInSection:(NSInteger)section {
    if (section == 0) {
#if PROSTO_INTERP
        NSString *mapping = @"未检测";
        switch (PSJitMemory) {
            case PSJitMemoryNotChecked:
                if (PSPreferredRuntime() == PSRuntimeInterpreter) mapping = @"已跳过（优先免 JIT）";
                break;
            case PSJitMemoryMapped:
                mapping = @"映射成功，尚未验证生成代码执行";
                break;
            case PSJitMemoryUnavailable:
                mapping = [NSString stringWithFormat:@"不可用（errno %d）", PSJitMemoryError];
                break;
            case PSJitMemoryUnsupported:
                mapping = @"当前环境不支持";
                break;
        }
        NSString *backend = @"免 JIT 解释器（实验）";
#else
        NSString *mapping = @"已跳过（此版本仅支持内置模组）";
        NSString *backend = @"此版本仅支持 IL2CPP 内置模组";
#endif
        return [NSString stringWithFormat:@"调试器：%@\nMAP_JIT：%@\n运行时：%@\nJIT 后端开发中。", PSAttachStatus(), mapping, backend];
    }
    if (section == 2) {
        NSString *result = [NSString stringWithContentsOfURL:[PSZipInbox() URLByAppendingPathComponent:@"result.txt"] encoding:NSUTF8StringEncoding error:NULL];
        NSString *help = @"进入游戏时安装 ZIP。在游戏模组菜单查看，或用“文件”管理 Mods 下的模组目录。";
        return result.length ? [help stringByAppendingFormat:@"\n%@",result] : help;
    }
    return nil;
}
- (UITableViewCell *)tableView:(UITableView *)table cellForRowAtIndexPath:(NSIndexPath *)path {
    UITableViewCell *cell = [[UITableViewCell alloc] initWithStyle:UITableViewCellStyleValue1 reuseIdentifier:nil];
    NSInteger row = path.row;
    if (path.section == 0) {
        cell.textLabel.text = @[@"自动",@"优先免 JIT 解释器",@"优先 JIT",@"重新检测"][row];
        if (row == PSPreferredRuntime()) cell.accessoryType = UITableViewCellAccessoryCheckmark;
        return cell;
    }
    if (path.section == 2 && row == 0) {
        cell.textLabel.text = @"导入模组 ZIP…";
        cell.accessoryType = UITableViewCellAccessoryDisclosureIndicator;
        return cell;
    }
    if (path.section == 2) {
        cell.textLabel.text = self.mods[row-1].lastPathComponent;
        cell.detailTextLabel.text = @"等待安装";
        cell.selectionStyle = UITableViewCellSelectionStyleNone;
        return cell;
    }
    NSString *identifier = @[@"item_details",@"smart_inventory_arrange"][row];
    cell.textLabel.text = @[@"Item Details",@"Smart Inventory Arrange"][row];
    NSString *key = PSModKey(identifier);
    NSNumber *value = [NSUserDefaults.standardUserDefaults objectForKey:key];
    // Older installs keep their ES3 preference until the user explicitly chooses.
    if (!value) cell.detailTextLabel.text = @"沿用游戏设置";
    else cell.detailTextLabel.text = value.boolValue ? @"已启用" : @"已停用";
    cell.accessoryType = UITableViewCellAccessoryDisclosureIndicator;
    return cell;
}
- (void)tableView:(UITableView *)table didSelectRowAtIndexPath:(NSIndexPath *)path {
    [table deselectRowAtIndexPath:path animated:YES];
    if (path.section == 0) {
        if (path.row <= PSRuntimeJit) {
            [NSUserDefaults.standardUserDefaults setInteger:path.row forKey:PSRuntimePreferenceKey];
        }
        PSRefreshJitMemory();
        [table reloadSections:[NSIndexSet indexSetWithIndex:0] withRowAnimation:UITableViewRowAnimationNone];
        return;
    }
    if (path.section == 2) {
        if (path.row != 0) return;
        UIDocumentPickerViewController *picker = [[UIDocumentPickerViewController alloc] initWithDocumentTypes:@[@"public.zip-archive"] inMode:UIDocumentPickerModeImport];
        picker.delegate = self;
        [self presentViewController:picker animated:YES completion:nil];
        return;
    }
    NSString *name = @[@"item_details",@"smart_inventory_arrange"][path.row];
    NSString *key = PSModKey(name);
    UIAlertController *menu = [UIAlertController alertControllerWithTitle:name message:nil preferredStyle:UIAlertControllerStyleAlert];
    for (NSString *label in @[@"启用",@"停用"]) {
        [menu addAction:[UIAlertAction actionWithTitle:label style:UIAlertActionStyleDefault handler:^(UIAlertAction *a) {
            [NSUserDefaults.standardUserDefaults setBool:[label isEqualToString:@"启用"] forKey:key];
            [self.tableView reloadData];
        }]];
    }
    [menu addAction:[UIAlertAction actionWithTitle:@"取消" style:UIAlertActionStyleCancel handler:nil]];
    [self presentViewController:menu animated:YES completion:nil];
}
- (void)documentPicker:(UIDocumentPickerViewController *)picker didPickDocumentsAtURLs:(NSArray<NSURL *> *)urls {
    NSURL *url = urls.firstObject;
    if (!url) return;
    NSError *error = nil;
    NSString *message = nil;
    if (!PSHasZipExtension(url)) message = @"请选择模组 ZIP。";
    else {
        NSFileManager *fm = NSFileManager.defaultManager;
        NSURL *folder = PSZipInbox();
        if (![fm createDirectoryAtURL:folder withIntermediateDirectories:YES attributes:nil error:&error] ||
            ![fm copyItemAtURL:url toURL:[folder URLByAppendingPathComponent:url.lastPathComponent] error:&error])
            message = error.localizedDescription;
    }
    [self refresh];
    if (message) {
        UIAlertController *alert = [UIAlertController alertControllerWithTitle:@"导入失败" message:message preferredStyle:UIAlertControllerStyleAlert];
        [alert addAction:[UIAlertAction actionWithTitle:@"好" style:UIAlertActionStyleDefault handler:nil]];
        dispatch_async(dispatch_get_main_queue(), ^{ [self presentViewController:alert animated:YES completion:nil]; });
    }
}
@end

@interface UnityAppController (PSUnityStartDeclaration)
- (void)startUnity:(UIApplication *)application;
@end

@interface PSLauncherAppController : UnityAppController
@property(nonatomic, strong) UIWindow *launcherWindow;
@property(nonatomic) BOOL gameStarted;
@end

@implementation PSLauncherAppController
- (void)startUnity:(UIApplication *)application {
    if (self.gameStarted || self.launcherWindow) return;
    PSLaunchSettings *settings = [[PSLaunchSettings alloc] initWithStyle:UITableViewStyleGrouped];
    UIWindow *window;
    if (@available(iOS 13.0, *)) {
        UIWindowScene *scene = self.window.windowScene;
        window = scene ? [[UIWindow alloc] initWithWindowScene:scene] : [[UIWindow alloc] initWithFrame:UIScreen.mainScreen.bounds];
    } else window = [[UIWindow alloc] initWithFrame:UIScreen.mainScreen.bounds];
    self.launcherWindow = window;
    window.rootViewController = [[UINavigationController alloc] initWithRootViewController:settings];
    window.windowLevel = UIWindowLevelNormal + 1;
    __weak PSLauncherAppController *weakSelf = self;
    settings.play = ^{ [weakSelf launchGame:application]; };
    [window makeKeyAndVisible];
}
- (void)launchGame:(UIApplication *)application {
    if (self.gameStarted) return;
    PSRefreshJitMemory();
    self.gameStarted = YES;
    self.launcherWindow.hidden = YES;
    self.launcherWindow = nil;
    [self.window makeKeyAndVisible];
    [super startUnity:application];
}
@end
IMPL_APP_CONTROLLER_SUBCLASS(PSLauncherAppController)
